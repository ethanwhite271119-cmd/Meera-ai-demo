import os
import json
import datetime
import webbrowser
import speech_recognition as sr
import pyttsx3

MEMORY_FILE = "meera_memory.json"

# ==========================================
# 1. MEMORY ENGINE (Long-Term & Short-Term)
# ==========================================
class MemoryEngine:
    def __init__(self, filepath=MEMORY_FILE):
        self.filepath = filepath
        self.long_term_data = self._load_memory()
        self.short_term_history = []  # In-session context buffer

    def _load_memory(self):
        if os.path.exists(self.filepath):
            try:
                with open(self.filepath, "r", encoding="utf-8") as f:
                    return json.load(f)
            except Exception:
                return {}
        return {
            "user_name": "Friend",
            "preferences": {},
            "learned_facts": []
        }

    def save_memory(self):
        with open(self.filepath, "w", encoding="utf-8") as f:
            json.dump(self.long_term_data, f, indent=4)

    def remember_fact(self, fact: str):
        self.long_term_data.setdefault("learned_facts", []).append(fact)
        self.save_memory()

    def set_user_name(self, name: str):
        self.long_term_data["user_name"] = name
        self.save_memory()

    def record_turn(self, role: str, content: str):
        self.short_term_history.append({"role": role, "content": content})
        # Keep only the last 10 conversational turns in RAM
        if len(self.short_term_history) > 10:
            self.short_term_history.pop(0)

# ==========================================
# 2. VOICE SYNTHESIS & RECOGNITION
# ==========================================
class MeeraVoice:
    def __init__(self):
        self.engine = pyttsx3.init()
        self.engine.setProperty("rate", 185)   # Natural speaking speed
        self.engine.setProperty("volume", 1.0)
        
        # Select a natural female or default voice if available
        voices = self.engine.getProperty("voices")
        for voice in voices:
            if "female" in voice.name.lower() or "zira" in voice.name.lower():
                self.engine.setProperty("voice", voice.id)
                break

        self.recognizer = sr.Recognizer()
        self.recognizer.dynamic_energy_threshold = True

    def speak(self, text: str):
        print(f"\n[Meera]: {text}")
        self.engine.say(text)
        self.engine.runAndWait()

    def listen(self) -> str:
        with sr.Microphone() as source:
            print("\n[Listening...]")
            self.recognizer.adjust_for_ambient_noise(source, duration=0.6)
            try:
                audio = self.recognizer.listen(source, timeout=6, phrase_time_limit=8)
                print("[Processing speech...]")
                query = self.recognizer.recognize_google(audio)
                print(f"[You]: {query}")
                return query.lower()
            except sr.WaitTimeoutError:
                return ""
            except sr.UnknownValueError:
                return ""
            except sr.RequestError:
                self.speak("Network connection error. Check your internet connection.")
                return ""

# ==========================================
# 3. AUTOMATION & COMMAND HANDLER
# ==========================================
class MeeraAssistant:
    def __init__(self):
        self.voice = MeeraVoice()
        self.memory = MemoryEngine()

    def greet(self):
        hour = datetime.datetime.now().hour
        user = self.memory.long_term_data.get("user_name", "Friend")
        greeting = "Good morning" if hour < 12 else "Good afternoon" if hour < 18 else "Good evening"
        self.voice.speak(f"{greeting}, {user}. Meera online and ready for your commands.")

    def handle_command(self, query: str) -> bool:
        """Processes the query. Returns False to terminate the loop."""
        self.memory.record_turn("user", query)

        # Shutdown / Exit
        if any(w in query for w in ["exit", "quit", "goodbye meera", "sleep meera"]):
            self.voice.speak("Powering down. Have a productive day ahead.")
            return False

        # Memory commands: Learn user name
        elif "my name is" in query:
            name = query.split("my name is")[-1].strip().capitalize()
            self.memory.set_user_name(name)
            self.voice.speak(f"Got it. I will address you as {name} from now on.")

        # Memory commands: Remember facts
        elif "remember that" in query:
            fact = query.split("remember that")[-1].strip()
            self.memory.remember_fact(fact)
            self.voice.speak(f"Committed to memory: {fact}")

        # Memory commands: Query facts
        elif "what do you remember" in query or "recall memory" in query:
            facts = self.memory.long_term_data.get("learned_facts", [])
            if facts:
                recent_facts = ", ".join(facts[-3:])
                self.voice.speak(f"Here are some facts stored in my memory: {recent_facts}")
            else:
                self.voice.speak("I don't have any facts stored in my long-term memory yet.")

        # System: Time and Date
        elif "time" in query:
            current_time = datetime.datetime.now().strftime("%I:%M %p")
            self.voice.speak(f"It is currently {current_time}.")

        elif "date" in query:
            today = datetime.datetime.now().strftime("%A, %B %d, %Y")
            self.voice.speak(f"Today is {today}.")

        # Web Automation
        elif "open youtube" in query:
            self.voice.speak("Launching YouTube.")
            webbrowser.open("https://www.youtube.com")

        elif "open google" in query:
            self.voice.speak("Opening Google search.")
            webbrowser.open("https://www.google.com")

        elif "search for" in query or "google" in query:
            search_term = query.replace("search for", "").replace("google", "").strip()
            self.voice.speak(f"Searching the web for {search_term}.")
            webbrowser.open(f"https://www.google.com/search?q={search_term}")

        # Catch-all fallback
        else:
            self.voice.speak(f"I heard: {query}. To perform deeper reasoning on that, connect my API key module.")

        return True

    def run(self):
        self.greet()
        active = True
        while active:
            query = self.voice.listen()
            if query:
                active = self.handle_command(query)

if __name__ == "__main__":
    meera = MeeraAssistant()
    meera.run()
